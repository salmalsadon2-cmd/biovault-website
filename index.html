Developer SDK integration

import { BioVaultAuth } from '@biovault/sdk';

// Initialize Account Abstraction Wallet
const wallet = new BioVaultAuth({
    network: 'mainnet',
    token: 'BVT'
});

// Prompt Biometric Signature
async function signTransaction(txData) {
    const signature = await wallet.requestBiometrics();
    if (signature.verified) {
        return await wallet.execute(txData);
    }
}
